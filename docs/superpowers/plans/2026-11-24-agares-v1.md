# Agares v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A menu-bar macOS app that types Vietnamese Telex in any app with no underlined composing text.

**Architecture:** A pure Swift library (`AgaresEngine`) turns typed letters into "delete N, type S" edits for the word under the cursor. A thin AppKit executable (`Agares`) hooks the keyboard with a `CGEventTap`, feeds the engine, and replays its edits as synthetic key events. Spec: `docs/superpowers/specs/2026-11-23-agares-design.md`.

**Tech Stack:** Swift 6 toolchain (language mode 5), SwiftPM, AppKit, CoreGraphics event taps, ServiceManagement, Swift Testing. No third-party dependencies.

```swift
/// A change to the text before the cursor: press Backspace `backspaces` times, then type `text`.
public struct Edit: Equatable {
    public let backspaces: Int
    public let text: String
}

extension Edit {
    /// Deletes back to the first character that differs, then types the rest.
    init(from old: String, to new: String) {
        let common = zip(old, new).prefix { $0.0 == $0.1 }.count
        self.init(backspaces: old.count - common, text: String(new.dropFirst(common)))
    }
}
```

```swift
enum Mark {
    case none, circumflex, breve, horn, stroke
}

/// Raw value is the column in `toneRows`.
enum Tone: Int {
    case acute = 1, grave, hook, tilde, dot
}

extension Character {
    var isVowel: Bool { "aeiouy".contains(self) }
}

struct Letter: Equatable {
    /// Lowercase ASCII.
    var base: Character
    var mark = Mark.none
    var upper = false
    /// A ư typed as a bare w. Undo turns it back into the w it was.
    var fromW = false

    var isVowel: Bool { base.isVowel }

    /// The lowercase letter with its mark and no tone.
    var marked: Character {
        switch (base, mark) {
        case ("a", .circumflex): return "â"
        case ("a", .breve): return "ă"
        case ("e", .circumflex): return "ê"
        case ("o", .circumflex): return "ô"
        case ("o", .horn): return "ơ"
        case ("u", .horn): return "ư"
        case ("d", .stroke): return "đ"
        default: return base
        }
    }

    /// The letter as shown: mark and case applied, plus the tone if this letter carries it.
    func rendered(tone: Tone?) -> String {
        var shown = marked
        if let tone, let row = toneRows.first(where: { $0[0] == shown }) {
            shown = row[tone.rawValue]
        }
        return upper ? shown.uppercased() : String(shown)
    }
}

/// Each row: no tone, sắc, huyền, hỏi, ngã, nặng.
private let toneRows: [[Character]] = [
    "aáàảãạ", "ăắằẳẵặ", "âấầẩẫậ", "eéèẻẽẹ", "êếềểễệ", "iíìỉĩị",
    "oóòỏõọ", "ôốồổỗộ", "ơớờởỡợ", "uúùủũụ", "ưứừửữự", "yýỳỷỹỵ",
].map(Array.init)
```

```swift
/// How a word splits into initial consonant, vowel group and final consonant.
struct Shape {
    /// Indices of the vowel group. Leaves out the u of "qu" and the i of "gi" before a vowel.
    let vowels: Range<Int>
    /// Judged on base letters only, ignoring marks.
    let isValid: Bool
    /// Final c, ch, p or t: such a syllable takes only sắc or nặng.
    let endsInStop: Bool
    /// The vowel that carries the tone, nil when the word has no vowel.
    let toneIndex: Int?

    /// `allowsZFWJ` also accepts z, f, w and j as initial consonants.
    init(_ letters: [Letter], allowsZFWJ: Bool = false) {
        let bases = letters.map(\.base)
        var i = bases.firstIndex(where: \.isVowel) ?? bases.count
        if bases.starts(with: "qu") || (bases.starts(with: "gi") && bases.count > 2 && bases[2].isVowel) {
            i = 2
        }
        let start = i
        while i < bases.count, bases[i].isVowel { i += 1 }
        let end = i
        while i < bases.count, !bases[i].isVowel { i += 1 }

        let initial = String(bases[..<start])
        let group = String(bases[start..<end])
        let final = String(bases[end..<i])
        vowels = start..<end
        isValid = i == bases.count
            && (initial.isEmpty || initials.contains(initial) || (allowsZFWJ && ["z", "f", "w", "j"].contains(initial)))
            && groups.contains(group)
            && (final.isEmpty || finals.contains(final))
        endsInStop = ["c", "ch", "p", "t"].contains(final)

        if let marked = vowels.last(where: { letters[$0].mark != .none }) {
            toneIndex = marked
        } else {
            switch vowels.count {
            case 0: toneIndex = nil
            case 1: toneIndex = start
            case 2: toneIndex = end < letters.count ? start + 1 : start
            default: toneIndex = start + 1
            }
        }
    }
}

private let initials: Set<String> = [
    "b", "c", "ch", "d", "g", "gh", "gi", "h", "k", "kh", "l", "m", "n", "ng", "ngh", "nh",
    "p", "ph", "qu", "r", "s", "t", "th", "tr", "v", "x",
]

private let groups: Set<String> = [
    "a", "e", "i", "o", "u", "y",
    "ai", "ao", "au", "ay", "eo", "eu", "ia", "ie", "iu", "oa", "oe", "oi", "oo",
    "ua", "ue", "ui", "uo", "uu", "uy", "ye",
    "ieu", "oai", "oao", "oay", "oeo", "uay", "uoi", "uou", "uya", "uye", "uyu", "yeu",
]

private let finals: Set<String> = ["c", "ch", "m", "n", "ng", "nh", "p", "t"]

/// Every vowel group Vietnamese writes with a circumflex. A doubling vowel adds one only if it makes one of these.
let circumflexGroups: Set<String> = [
    "â", "ê", "ô",
    "âu", "ây", "êu", "iê", "ôi", "uâ", "uê", "uô", "yê",
    "iêu", "yêu", "uây", "uôi", "uyê",
]

/// Every vowel group Vietnamese writes with a horn or a breve. A w adds one only if it makes one of these.
let hornGroups: Set<String> = [
    "ă", "ơ", "ư",
    "oă", "ơi", "uơ", "ưa", "ưi", "ưu", "ươ",
    "ươi", "ươu",
]
```

### Task 4: The Telex engine

**Files:**

- Create: `Sources/AgaresEngine/Engine.swift`
- Test: `Tests/AgaresEngineTests/EngineTests.swift`

**Interfaces:**

- Consumes: `Edit(from:to:)` from Task 1; `Letter`, `Mark`, `Tone` from Task 2; `Shape` from Task 3.
- Produces:
  - `public struct Engine` with `public init(allowsZFWJ: Bool = false)`
  - `public mutating func type(_ key: Character) -> Edit?` — `nil` means the key should reach the app unchanged
  - `public mutating func backspace() -> Edit?` — `nil` means Backspace should reach the app unchanged
  - `public mutating func reset()`

Background for the implementer: the engine holds the word as letters plus one tone for the whole word. After each key it renders the word and diffs it against the previous rendering. The rules are spec section 4; the table in the test below is the authority. Three things are easy to get wrong:

- A doubling vowel (`a`, `e`, `o`) marks an earlier vowel of the same letter, however many letters came in between, whenever that makes one of `circumflexGroups`: `lay` plus `a` is lây. Otherwise it is a new vowel, which is why `ngoeo` stays `ngoeo` (ôe is not a group).
- `w` likewise marks a vowel only when that makes one of `hornGroups`, so `ai` plus `w` stays `aiw`.
- A vowel group holds one kind of mark: circumflex, or the marks `w` makes (horn and breve), never both.
- After a mark key is undone (`aaa`, `ass`), that key is a plain letter for the rest of the word.

- [ ] **Step 1: Write the failing test**

Write `Tests/AgaresEngineTests/EngineTests.swift`:

```swift
import Testing
@testable import AgaresEngine

/// Plays keys into a fresh engine and returns what a text field would show. "⌫" is Backspace.
func typed(_ keys: String, allowsZFWJ: Bool = false) -> String {
    var engine = Engine(allowsZFWJ: allowsZFWJ)
    var field = ""
    for key in keys {
        let isBackspace = key == "⌫"
        if let edit = isBackspace ? engine.backspace() : engine.type(key) {

    }
    return field
```

```swift
/// Telex state for the word under the cursor.
public struct Engine {
    private var letters: [Letter] = []
    private var tone: Tone?
    /// Mark keys undone in this word. They stay plain letters until the word ends.
    private var spent: Set<Character> = []
    private var shown = ""
    private let allowsZFWJ: Bool

    /// `allowsZFWJ` accepts z, f, w and j as initial consonants, so words such as zô and jì take marks.
    public init(allowsZFWJ: Bool = false) {
        self.allowsZFWJ = allowsZFWJ
    }

    /// Feeds one ASCII letter. Returns nil when the key should reach the app unchanged.
    public mutating func type(_ key: Character) -> Edit? {
        guard key.isASCII, key.isLetter else {
            reset()
            return nil
        }
        apply(key)
        completeHornPair()
        let old = shown
        shown = render()
        return shown == old + String(key) ? nil : Edit(from: old, to: shown)
    }

    /// Returns nil when Backspace should reach the app unchanged.
    public mutating func backspace() -> Edit? {
        guard !letters.isEmpty else { return nil }
        // Deleting the vowel that carries the tone deletes the tone with it.
        if parse(letters).toneIndex == letters.count - 1 { tone = nil }
        // Deleting the plain letter that an accidental repeat left behind makes that key a mark key again.
        spent.remove(letters.removeLast().base)
        guard !letters.isEmpty else {
            reset()
            return nil
        }
        let old = shown
        shown = render()
        return shown == String(old.dropLast()) ? nil : Edit(from: old, to: shown)
    }

    /// The word has ended.
    public mutating func reset() {
        self = Engine(allowsZFWJ: allowsZFWJ)
    }

    private func parse(_ letters: [Letter]) -> Shape {
        Shape(letters, allowsZFWJ: allowsZFWJ)
    }

    private func render() -> String {
        let toneIndex = tone == nil ? nil : parse(letters).toneIndex
        return letters.indices.map { letters[$0].rendered(tone: $0 == toneIndex ? tone : nil) }.joined()
    }

    private mutating func apply(_ key: Character) {
        let lower = Character(key.lowercased())
        if !spent.contains(lower) {
            switch lower {
            case "a", "e", "o":
                if circumflex(lower, typed: key) { return }
            case "w":
                if horn(typed: key) { return }
            case "d":
                if stroke(typed: key) { return }
            case "z":
                if tone != nil {
                    tone = nil
                    return
                }
            default:
                if let new = toneKeys[lower], setTone(new, typed: key) { return }
            }
        }
        append(key)
    }

    private mutating func append(_ key: Character) {
        letters.append(Letter(base: Character(key.lowercased()), upper: key.isUppercase))
    }

    /// The key becomes a plain letter, now and for the rest of the word.
    private mutating func undo(_ key: Character) {
        spent.insert(Character(key.lowercased()))
        append(key)
    }

    /// Spec 4.5: a valid shape, and a stop final (c, ch, p, t) only with sắc, nặng or no tone.
    private func accepts(_ tone: Tone?, _ shape: Shape) -> Bool {
        shape.isValid && (!shape.endsInStop || tone == nil || tone == .acute || tone == .dot)
    }

    // MARK: aa ee oo

    private mutating func circumflex(_ lower: Character, typed key: Character) -> Bool {
        let shape = parse(letters)
        guard accepts(tone, shape) else { return false }
        let sameVowel = shape.vowels.filter { letters[$0].base == lower }

        // A second press takes the circumflex off again.
        if let marked = sameVowel.first(where: { letters[$0].mark == .circumflex }) {
            letters[marked].mark = .none
            undo(key)
            return true
        }
        // The key marks an earlier vowel wherever that makes a vowel group Vietnamese has, however many
        // letters came in between: lay + a is lây. It is a new vowel otherwise, as in ngoeo, where ôe is no group.
        for target in sameVowel {
            var marked = letters
            marked[target].mark = .circumflex
            if circumflexGroups.contains(String(marked[shape.vowels].map(\.marked))) {
                letters = marked
                return true
            }
        }
        return false
    }

    // MARK: w

    private mutating func horn(typed key: Character) -> Bool {
        let shape = parse(letters)
        let group = shape.vowels
        if group.isEmpty {
            // With z, f, w and j allowed as consonants, a w that starts the word is one of them.
            if allowsZFWJ, letters.isEmpty { return false }
            // Otherwise a bare w is a new ư, if that leaves a valid syllable.
            let bare = Letter(base: "u", mark: .horn, upper: key.isUppercase, fromW: true)
            guard parse(letters + [bare]).isValid else { return false }
            letters.append(bare)
            return true
        }
        guard accepts(tone, shape) else { return false }

        let targets = hornTargets(in: group)
        if !targets.isEmpty {
            for i in targets { letters[i].mark = letters[i].base == "a" ? .breve : .horn }
        } else if group.contains(where: { letters[$0].mark == .horn || letters[$0].mark == .breve }) {
            undoHorn(typed: key)
        } else {
            return false
        }
        return true
    }

    /// The vowels a w would newly mark: the first choice, in the priority order of spec 4.1, that
    /// makes a vowel group Vietnamese has. Empty when there is none.
    private func hornTargets(in group: Range<Int>) -> [Int] {
        func plain(_ i: Int) -> Bool { letters[i].mark == .none }
        func makesGroup(_ targets: [Int]) -> Bool {
            var marked = letters
            for i in targets { marked[i].mark = marked[i].base == "a" ? .breve : .horn }
            return !targets.isEmpty && hornGroups.contains(String(marked[group].map(\.marked)))
        }

        var choices: [[Int]] = []
        for i in group.dropLast() where letters[i].base == "u" {
            let next = i + 1
            if letters[next].base == "o" {
                // uo becomes uơ at the end of the word, ươ once a letter follows. Nothing else is tried.
                var pair = plain(next) ? [next] : []
                if plain(i), next < letters.count - 1 { pair.insert(i, at: 0) }
                return makesGroup(pair) ? pair : []
            }
            if letters[next].base == "a", plain(i) { choices.append([i]) }
        }
        for base in ["a", "o", "u"] as [Character] {
            choices += group.filter { letters[$0].base == base && plain($0) }.map { [$0] }
        }
        return choices.first(where: makesGroup) ?? []
    }

    private mutating func undoHorn(typed key: Character) {
        spent.insert("w")
        let endsWithBareW = letters.last?.fromW == true
        for i in letters.indices where letters[i].mark == .horn || letters[i].mark == .breve {
            if letters[i].fromW {
                letters[i] = Letter(base: "w", upper: letters[i].upper)
            } else {
                letters[i].mark = .none
            }
        }
        // "ww" is one w: the ư goes back to being the first w, and the second is dropped.
        if !endsWithBareW { append(key) }
    }

    /// uơ and ưo become ươ once a letter follows: mươn, lương, người.
    private mutating func completeHornPair() {
        let group = parse(letters).vowels
        for i in group.dropLast() where letters[i].base == "u" && letters[i + 1].base == "o" && i + 1 < letters.count - 1 {
            let marks = (letters[i].mark, letters[i + 1].mark)
            if marks == (.horn, .none) || marks == (.none, .horn) {
                letters[i].mark = .horn
                letters[i + 1].mark = .horn
            }
        }
    }

    // MARK: dd

    private mutating func stroke(typed key: Character) -> Bool {
        guard let first = letters.first, first.base == "d" else { return false }
        if first.mark == .stroke {
            letters[0].mark = .none
            undo(key)
        } else if letters.count == 1 || accepts(tone, parse(letters)) {
            letters[0].mark = .stroke
        } else {
            return false
        }
        return true
    }

    // MARK: s f r x j

    private mutating func setTone(_ new: Tone, typed key: Character) -> Bool {
        let shape = parse(letters)
        guard !shape.vowels.isEmpty else { return false }
        if tone == new {
            tone = nil
            undo(key)
            return true
        }
        guard accepts(new, shape) else { return false }
        tone = new
        return true
    }
}

private let toneKeys: [Character: Tone] = ["s": .acute, "f": .grave, "r": .hook, "x": .tilde, "j": .dot]
```

```swift
import AgaresEngine
import AppKit

/// Replays an Edit in the focused app as synthetic key events.
final class Typist {
    /// Marks our own events so KeyTap lets them through.
    static let tag: Int64 = 0x4147_4152

    // A private source keeps physically held modifiers (Shift while typing capitals) off our events.
    private let source = CGEventSource(stateID: .privateState)

    func send(_ edit: Edit, through proxy: CGEventTapProxy) {
        if edit.backspaces > 0 {
            if spotlightIsOpen() {
                // Select the old characters; the text typed next replaces them.
                for _ in 0..<edit.backspaces { press(123, flags: .maskShift, proxy) }
            } else {
                // An invisible character first: in a field with inline autocomplete it replaces the
                // highlighted suggestion, so the backspaces then delete real text.
                type("\u{202F}", proxy)
                for _ in 0...edit.backspaces { press(51, flags: [], proxy) }
            }
        }
        type(edit.text.precomposedStringWithCanonicalMapping, proxy)
    }

    private func press(_ keyCode: CGKeyCode, flags: CGEventFlags, _ proxy: CGEventTapProxy) {
        for keyDown in [true, false] {
            guard let event = CGEvent(keyboardEventSource: source, virtualKey: keyCode, keyDown: keyDown) else { continue }
            event.flags.formUnion(flags)
            post(event, proxy)
        }
    }

    private func type(_ text: String, _ proxy: CGEventTapProxy) {
        guard !text.isEmpty else { return }
        let units = Array(text.utf16)
        for keyDown in [true, false] {
            guard let event = CGEvent(keyboardEventSource: source, virtualKey: 0, keyDown: keyDown) else { continue }
            event.keyboardSetUnicodeString(stringLength: units.count, unicodeString: units)
            post(event, proxy)
        }
    }

    // Posting through the tap keeps our events in order with the keys still queued behind them.
    private func post(_ event: CGEvent, _ proxy: CGEventTapProxy) {
        event.setIntegerValueField(.eventSourceUserData, value: Typist.tag)
        event.tapPostEvent(proxy)
    }

    /// The last answer of `spotlightIsOpen`, and when it was found. Asking macOS for its windows costs
    /// about a millisecond, far too much for every keystroke, so the answer is reused.
    private var spotlightOpen: Bool?
    private var spotlightCheckedAt = 0.0

    /// Call when Spotlight may have opened or closed: a shortcut, an unusual key, a click, an app switch.
    func forgetSpotlight() {
        spotlightOpen = nil
    }

    private func spotlightIsOpen() -> Bool {
        let now = CFAbsoluteTimeGetCurrent()
        // The two seconds cover ways of opening Spotlight that the key hook never sees.
        if let spotlightOpen, now - spotlightCheckedAt < 2 { return spotlightOpen }
        let open = lookForSpotlight()
        spotlightOpen = open
        spotlightCheckedAt = now
        return open
    }

    private func lookForSpotlight() -> Bool {
        if NSWorkspace.shared.frontmostApplication?.bundleIdentifier == "com.apple.Spotlight" { return true }
        let windows = CGWindowListCopyWindowInfo([.optionOnScreenOnly, .excludeDesktopElements], kCGNullWindowID)
        return (windows as? [[String: Any]] ?? []).contains { $0[kCGWindowOwnerName as String] as? String == "Spotlight" }
    }
}
```

```swift
import AgaresEngine
import AppKit

/// The system-wide key hook. Everything it does not handle passes through untouched.
final class KeyTap {
    var vietnameseOn = true {
        didSet { engine.reset() }
    }

    /// Called when Ctrl+Shift is pressed and released with no other key in between.
    var onHotkey: () -> Void = {}

    private var engine = Engine()
    private let typist = Typist()
    private var tap: CFMachPort?
    private var hotkeyArmed = false

    init() {
        NSWorkspace.shared.notificationCenter.addObserver(
            forName: NSWorkspace.didActivateApplicationNotification, object: nil, queue: .main
        ) { [weak self] _ in self?.engine.reset() }
    }

    /// False when macOS refuses the tap, which means Accessibility is not granted.
    func start() -> Bool {
        let types: [CGEventType] = [.keyDown, .flagsChanged, .leftMouseDown, .rightMouseDown, .otherMouseDown]
        let mask = types.reduce(CGEventMask(0)) { $0 | CGEventMask(1) << $1.rawValue }
        guard let tap = CGEvent.tapCreate(
            tap: .cgSessionEventTap, place: .headInsertEventTap, options: .defaultTap,
            eventsOfInterest: mask,
            callback: { proxy, type, event, refcon in
                Unmanaged<KeyTap>.fromOpaque(refcon!).takeUnretainedValue().handle(proxy, type, event)
            },
            userInfo: Unmanaged.passUnretained(self).toOpaque()
        ) else { return false }
        CFRunLoopAddSource(CFRunLoopGetMain(), CFMachPortCreateRunLoopSource(nil, tap, 0), .commonModes)
        CGEvent.tapEnable(tap: tap, enable: true)
        self.tap = tap
        return true
    }

    func stop() {
        guard let tap else { return }
        CFMachPortInvalidate(tap)
        self.tap = nil
        engine.reset()
    }

    private func handle(_ proxy: CGEventTapProxy, _ type: CGEventType, _ event: CGEvent) -> Unmanaged<CGEvent>? {
        let pass = Unmanaged.passUnretained(event)
        switch type {
        case .tapDisabledByTimeout, .tapDisabledByUserInput:
            if let tap { CGEvent.tapEnable(tap: tap, enable: true) }
            return pass
        case .flagsChanged:
            modifiersChanged(event.flags)
            return pass
        case .keyDown:
            break
        default:
            // A mouse button: the cursor may have moved.
            hotkeyArmed = false
            engine.reset()
            return pass
        }

        if event.getIntegerValueField(.eventSourceUserData) == Typist.tag { return pass }
        hotkeyArmed = false
        if !vietnameseOn || !event.flags.isDisjoint(with: [.maskCommand, .maskControl, .maskAlternate]) {
            engine.reset()
            return pass
        }

        let edit: Edit?
        if event.getIntegerValueField(.keyboardEventKeycode) == 51 {
            edit = engine.backspace()
        } else if let letter = Self.letter(of: event) {
            edit = engine.type(letter)
        } else {
            engine.reset()
            return pass
        }
        guard let edit else { return pass }
        typist.send(edit, through: proxy)
        return nil
    }

    private func modifiersChanged(_ flags: CGEventFlags) {
        let held = flags.intersection([.maskControl, .maskShift, .maskCommand, .maskAlternate])
        if held == [.maskControl, .maskShift] {
            hotkeyArmed = true
            return
        }
        // Still armed here means no key or click came between press and release.
        if hotkeyArmed, held.isSubset(of: [.maskControl, .maskShift]) { onHotkey() }
        hotkeyArmed = false
    }

    /// The typed character if it is a single ASCII letter, with Shift and Caps Lock already applied.
    private static func letter(of event: CGEvent) -> Character? {
        var length = 0
        var units = [UniChar](repeating: 0, count: 4)
        event.keyboardGetUnicodeString(maxStringLength: 4, actualStringLength: &length, unicodeString: &units)
        guard length == 1, let scalar = Unicode.Scalar(units[0]) else { return nil }
        let character = Character(scalar)
        return character.isASCII && character.isLetter ? character : nil
    }
}
```

```swift
    }

    private func refreshTitle() {
        statusItem.button?.title = !tapRunning ? "⚠" : keyTap.vietnameseOn ? "V" : "E"
    }

    func menuNeedsUpdate(_ menu: NSMenu) {
        menu.removeAllItems()
        if !tapRunning {
            let settings = menu.addItem(
                withTitle: "Open Accessibility Settings…", action: #selector(openAccessibilitySettings), keyEquivalent: "")
            settings.target = self
            menu.addItem(.separator())
        }
        let vietnamese = menu.addItem(withTitle: "Vietnamese", action: #selector(toggleVietnamese), keyEquivalent: "")
        vietnamese.target = self
        vietnamese.state = keyTap.vietnameseOn ? .on : .off
        let login = menu.addItem(withTitle: "Launch at login", action: #selector(toggleLaunchAtLogin), keyEquivalent: "")
        login.target = self
        login.state = SMAppService.mainApp.status == .enabled ? .on : .off
        menu.addItem(.separator())
        menu.addItem(withTitle: "Quit Agares", action: #selector(NSApplication.terminate(_:)), keyEquivalent: "q")
    }

    @objc private func toggleVietnamese() {
        keyTap.vietnameseOn.toggle()
        UserDefaults.standard.set(keyTap.vietnameseOn, forKey: Self.onKey)
        refreshTitle()
    }

    @objc private func toggleLaunchAtLogin() {
        do {
            if SMAppService.mainApp.status == .enabled {
                try SMAppService.mainApp.unregister()
            } else {
                try SMAppService.mainApp.register()
            }
        } catch {
            NSAlert(error: error).runModal()
        }
    }

    @objc private func openAccessibilitySettings() {
        NSWorkspace.shared.open(URL(string: "x-apple.systempreferences:com.apple.preference.security?Privacy_Accessibility")!)
    }
}
```

- [ ] **Step 2: Write the certificate script**

Write `scripts/make-cert.sh`:

```bash
#!/bin/bash
# One-time: creates the self-signed "Agares Dev" code-signing certificate in the login keychain.
set -euo pipefail

NAME="Agares Dev"
KEYCHAIN="$HOME/Library/Keychains/login.keychain-db"

if security find-certificate -c "$NAME" "$KEYCHAIN" >/dev/null 2>&1; then
  echo "Certificate \"$NAME\" already exists."
  exit 0
fi

TMP=$(mktemp -d)
trap 'rm -rf "$TMP"' EXIT

cat > "$TMP/cert.conf" <<EOF
[req]
distinguished_name = dn
x509_extensions = ext
prompt = no
[dn]
CN = $NAME
[ext]
basicConstraints = critical,CA:false
keyUsage = critical,digitalSignature
extendedKeyUsage = critical,codeSigning
EOF

# /usr/bin/openssl is LibreSSL, whose PKCS#12 output the security tool can read.
/usr/bin/openssl req -x509 -newkey rsa:2048 -nodes -days 3650 \
  -config "$TMP/cert.conf" -keyout "$TMP/key.pem" -out "$TMP/cert.pem"
/usr/bin/openssl pkcs12 -export -inkey "$TMP/key.pem" -in "$TMP/cert.pem" \
  -name "$NAME" -out "$TMP/identity.p12" -passout pass:agares
security import "$TMP/identity.p12" -k "$KEYCHAIN" -P agares -T /usr/bin/codesign

echo "Created \"$NAME\"."
```

Write `scripts/bundle.sh`:

```bash
#!/bin/bash
# Builds build/Agares.app and signs it.
#   bundle.sh install   also copies it to /Applications and launches it
#   bundle.sh dmg       also writes build/Agares-<version>.dmg
# The release workflow sets SWIFT_BUILD_FLAGS for a universal binary.
set -euo pipefail
cd "$(dirname "$0")/.."

APP=build/Agares.app
ICONSET=build/AppIcon.iconset
# shellcheck disable=SC2206  # SWIFT_BUILD_FLAGS is split into words on purpose
FLAGS=(-c release ${SWIFT_BUILD_FLAGS:-})

swift build "${FLAGS[@]}"

rm -rf "$APP" "$ICONSET"
mkdir -p "$APP/Contents/MacOS" "$APP/Contents/Resources" "$ICONSET"
cp "$(swift build "${FLAGS[@]}" --show-bin-path)/Agares" "$APP/Contents/MacOS/Agares"
cp Resources/Info.plist "$APP/Contents/Info.plist"

for size in 16 32 128 256 512; do
  rsvg-convert -w "$size" -h "$size" Resources/icon.svg -o "$ICONSET/icon_${size}x${size}.png"
  rsvg-convert -w $((size * 2)) -h $((size * 2)) Resources/icon.svg -o "$ICONSET/icon_${size}x${size}@2x.png"
done
iconutil -c icns "$ICONSET" -o "$APP/Contents/Resources/AppIcon.icns"

# Always the same certificate: macOS ties the Accessibility permission to the signer.
codesign --force --sign "Agares Dev" --identifier com.quyennguyenvu.agares "$APP"
codesign --verify --strict --verbose=2 "$APP"

case "${1:-}" in
  install)
    pkill -x Agares || true
    # LaunchServices keeps listing the old instance for a moment after its process is gone, and
    # "open" fails with error -600 or -609 until it stops. Waiting for the process alone is not enough.
    while [ -n "$(lsappinfo find bundleID=com.quyennguyenvu.agares)" ]; do sleep 0.1; done
    rm -rf /Applications/Agares.app
    cp -R "$APP" /Applications/Agares.app
    open /Applications/Agares.app
    ;;
  dmg)
    VERSION=$(plutil -extract CFBundleShortVersionString raw Resources/Info.plist)
    STAGE=build/dmg
    rm -rf "$STAGE"
    mkdir -p "$STAGE"
    cp -R "$APP" "$STAGE/"
    # The shortcut the user drags the app onto.
    ln -s /Applications "$STAGE/Applications"
    hdiutil create -volname Agares -srcfolder "$STAGE" -ov -format UDZO "build/Agares-$VERSION.dmg"
    ;;
esac
```
